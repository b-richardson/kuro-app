# Obsidian Template Creator Module System

This document describes how to create and register new modules for the Obsidian Template Creator.

## Overview

Modules are reusable components that allow users to build templates by combining different types of content (text, headers, dividers, etc.). Each module:

1. Manages its own UI and editing experience
2. Produces markdown output
3. Integrates with the module canvas system

## Module Architecture

### Key Components

- **BaseModule**: The core wrapper component that provides common functionality (selection, movement, duplicate, delete)
- **TemplateModuleEditor**: Maps module types to their component implementations
- **ModuleCanvas**: Renders all modules and manages selection state
- **Template Store**: Manages the module data, validation, and persistence

### Module Flow

When a module is added to the canvas:

1. The type is validated in `template.ts` (isValidModuleType)
2. Default content is generated by `pluginTemplatesUtils.ts`
3. The module is rendered through TemplateModuleEditor
4. The specific module component (e.g., DividerModule) handles the UI

## Creating a New Module

### Step 1: Register the Module Type

Add your new module type to the ModuleType definition in `/src/types/TemplateModule.ts`:

```typescript
export type ModuleType =
  // Basic module types
  | "header"
  | "checklist"
  | "table"
  | "text"
  | "divider"
  | "your-new-module" // Add your new type here
  // Plugin-specific module types
  | "yaml"
  | "templaterField"
  | "dataviewQuery";
```

### Step 2: Add the Module to the Template Store

Make sure the module type is included in the `isValidModuleType` function in `/src/stores/template.ts`:

```typescript
const isValidModuleType = (type: any): type is ModuleType => {
  const validTypes: ModuleType[] = [
    "header",
    "checklist",
    "table",
    "text",
    "divider",
    "your-new-module", // Add your new type here
    "yaml",
    "templaterField",
    "dataviewQuery",
    "kanban",
  ];
  return validTypes.includes(type as ModuleType);
};
```

### Step 3: Define Default Content

Add default content generation in `/src/utils/pluginTemplatesUtils.ts`:

```typescript
export function generateDefaultContent(
  moduleType: ModuleType,
  pluginType: PluginType = null
): string[] {
  switch (moduleType) {
    // ... existing cases

    case "your-new-module":
      return ["Default content for your module"];

    // ... other cases
  }

// Also add an icon in the same file

  switch (moduleType) {
    // ... existing cases

  }

```

### Step 4: Create the Module Component

Create a new file in the appropriate directory:

- Basic content modules go in `/src/components/TemplateCreator/modules/content/`
- Plugin-specific modules go in `/src/components/TemplateCreator/modules/plugins/`

**Example module template:**

```vue
<template>
  <BaseModule
    :module="module"
    :index="index"
    :modules="modules"
    :isSelected="isSelected"
    :autoFocus="autoFocus"
    @update:content="updateContent"
    @update="$emit('update', $event)"
    @delete="$emit('delete')"
    @select="$emit('select')"
    @duplicate="$emit('duplicate')"
    @module-change="$emit('module-change')"
    @add-line="handleAddLine"
    ref="baseModuleRef"
  >
    <!-- Your custom module UI goes here -->
    <div class="your-module-content">
      <!-- Example: content editing UI -->
      <div
        v-for="(line, lineIndex) in module.content"
        :key="lineIndex"
        class="your-module-line"
      >
        <input
          v-model="localContent[lineIndex]"
          @input="updateLocalContent(lineIndex)"
          class="your-module-input"
        />
      </div>
    </div>
  </BaseModule>
</template>

<script setup lang="ts">
import { ref, watch, onMounted } from "vue";
import BaseModule from "../core/BaseModule.vue";
import { TemplateModule } from "@/types/TemplateModule";

const props = defineProps<{
  module: TemplateModule;
  index: number;
  modules: TemplateModule[];
  isSelected?: boolean;
  autoFocus?: boolean;
}>();

const emit = defineEmits<{
  (e: "update:content", content: string[]): void;
  (e: "update", modules: TemplateModule[]): void;
  (e: "delete"): void;
  (e: "select"): void;
  (e: "duplicate"): void;
  (e: "module-change"): void;
}>();

// Reference to the base module component
const baseModuleRef = ref<InstanceType<typeof BaseModule> | null>(null);

// Local content for reactivity
const localContent = ref<string[]>([...props.module.content]);

// Watch for external content changes
watch(
  () => props.module.content,
  (newContent) => {
    localContent.value = [...newContent];
  },
  { deep: true }
);

// Update content in parent component
const updateContent = (content: string[]) => {
  emit("update:content", content);
};

// Update local content and emit changes
const updateLocalContent = (index: number) => {
  const newContent = [...localContent.value];
  emit("update:content", newContent);
};

// Handle adding a new line
const handleAddLine = () => {
  const newContent = [...localContent.value, "New line"];
  localContent.value = newContent;
  emit("update:content", newContent);
};

// Set editing state (example)
const setEditing = (state: boolean) => {
  baseModuleRef.value?.setEditing(state);
};

// When component is mounted
onMounted(() => {
  // Any initialization code
});
</script>

<style scoped>
@import "../core/moduleStyles.css";

/* Your module-specific styles here */
.your-module-content {
  /* Custom styling */
}

.your-module-line {
  margin-bottom: 8px;
}

.your-module-input {
  width: 100%;
  padding: 8px;
  border: 1px solid var(--color-surface);
  border-radius: 4px;
}
</style>
```

### Step 5: Register in TemplateModuleEditor

Update `/src/components/Editor/TemplateModuleEditor.vue` to include your module:

```typescript
// Import module components
import YourNewModule from "@/components/TemplateCreator/modules/content/YourNewModule.vue";

// And in the computed property:
const moduleComponent = computed(() => {
  switch (props.module.type) {
    // ... existing cases

    case "your-new-module":
      return YourNewModule;

    // ... other cases
  }
});
```

### Step 6: Add to Module Library

Add your module to the appropriate module library component. For basic modules, update `/src/components/ModuleLibrary/BasicModules.vue`:

```typescript
const basicModules = [
  // ... existing modules

  {
    type: "your-new-module",
    title: "Your Module Name",
    description: "Description of what this module does",
    priority: "high", // or "medium" or "low"
  },
];
```

## Module Types Guide

### Basic Content Modules

These modules represent basic markdown elements:

- **Header**: Section headings (H1-H6)
- **Text**: Paragraphs of plain text
- **Checklist**: List of checkable items
- **Table**: Data organized in rows and columns
- **Divider**: Horizontal rule separator (---)

### Plugin-Specific Modules

These modules integrate with Obsidian plugins:

- **YAML**: Frontmatter metadata
- **Templater**: Dynamic content using Templater plugin syntax
- **Dataview Query**: Queries using Dataview plugin
- **Dataview Fields**: Inline dataview fields
- **Kanban**: Kanban board using the Kanban plugin

## Best Practices

1. **Use BaseModule**: Always extend BaseModule for consistency
2. **Validate Types**: Ensure your module type is registered in all required locations
3. **Local State**: Use local reactive state for editing, then emit changes
4. **Handle Selection**: Properly manage selection and focus states
5. **Custom Styling**: Use scoped styles for your module's unique appearance
6. **Test Rendering**: Verify your module renders correctly in the Obsidian editor preview

## Example: DividerModule

The DividerModule is a simple example that creates a horizontal rule (---):

```vue
<template>
  <BaseModule
    :module="module"
    :index="index"
    :modules="modules"
    :isSelected="isSelected"
    :autoFocus="autoFocus"
    @update:content="updateContent"
    @update="$emit('update', $event)"
    @delete="$emit('delete')"
    @select="$emit('select')"
    @duplicate="$emit('duplicate')"
    @module-change="$emit('module-change')"
    @add-line="handleAddLine"
    ref="baseModuleRef"
  >
    <!-- Simple visual divider -->
    <div class="divider-visual">
      <i class="fas fa-minus divider-icon"></i>
      <hr class="divider-line" />
    </div>
  </BaseModule>
</template>

<script setup lang="ts">
import { ref } from "vue";
import BaseModule from "../core/BaseModule.vue";
import { TemplateModule } from "@/types/TemplateModule";

const props = defineProps<{
  module: TemplateModule;
  index: number;
  modules: TemplateModule[];
  isSelected?: boolean;
  autoFocus?: boolean;
}>();

const emit = defineEmits<{
  (e: "update:content", content: string[]): void;
  (e: "update", modules: TemplateModule[]): void;
  (e: "delete"): void;
  (e: "select"): void;
  (e: "duplicate"): void;
  (e: "module-change"): void;
}>();

// Reference to the base module component
const baseModuleRef = ref<InstanceType<typeof BaseModule> | null>(null);

// Ensure content is always "---"
const updateContent = (content: string[]) => {
  emit("update:content", ["---"]);
};

// No-op for add line - dividers should always be a single line
const handleAddLine = () => {
  // Do nothing - prevents adding additional lines
};
</script>
```

## Troubleshooting

1. **Module Not Appearing**: Check that your module type is registered in all required locations
2. **Wrong Module Rendering**: Verify the TemplateModuleEditor's switch statement
3. **Module Showing as Text**: Ensure the type is in the isValidModuleType function
4. **Styling Issues**: Check for CSS specificity problems; consider using :deep() for child selectors
